cmake_minimum_required(VERSION 3.20)
project(mediapipe_hands LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  # A DLL distribuída não exige instalar outro runtime C++ no PC do estudante.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_library(driver_mediapipe_hands SHARED src/driver.cpp)
target_include_directories(driver_mediapipe_hands SYSTEM PRIVATE third_party/openvr)
set_target_properties(driver_mediapipe_hands PROPERTIES PREFIX "")
if(WIN32)
  target_link_libraries(driver_mediapipe_hands PRIVATE ws2_32)
  set(DRIVER_PLATFORM win64)
else()
  set(DRIVER_PLATFORM linux64)
  target_compile_options(driver_mediapipe_hands PRIVATE -Wall -Wextra -Werror)
endif()
if(MSVC)
  target_compile_options(driver_mediapipe_hands PRIVATE /W4 /WX /permissive-)
endif()
set(DRIVER_OUTPUT "${CMAKE_BINARY_DIR}/package/mediapipe_hands")
file(COPY driver/mediapipe_hands/ DESTINATION "${DRIVER_OUTPUT}")
file(COPY third_party/openvr/LICENSE DESTINATION "${DRIVER_OUTPUT}/licenses/openvr")
set_target_properties(driver_mediapipe_hands PROPERTIES
  LIBRARY_OUTPUT_DIRECTORY "${DRIVER_OUTPUT}/bin/${DRIVER_PLATFORM}"
  RUNTIME_OUTPUT_DIRECTORY "${DRIVER_OUTPUT}/bin/${DRIVER_PLATFORM}"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${DRIVER_OUTPUT}/bin/${DRIVER_PLATFORM}")
add_executable(protocol_tests tests/protocol_tests.cpp)
add_test(NAME protocol COMMAND protocol_tests)
add_executable(receiver_tests tests/receiver_tests.cpp)
if(WIN32)
  target_link_libraries(receiver_tests PRIVATE ws2_32)
endif()
add_test(NAME receiver COMMAND receiver_tests)
add_executable(factory_tests tests/factory_tests.cpp)
target_include_directories(factory_tests SYSTEM PRIVATE third_party/openvr)
target_link_libraries(factory_tests PRIVATE driver_mediapipe_hands)
if(WIN32)
  # O loader Windows não usa o RPATH do Linux. O teste precisa encontrar a
  # DLL ao lado do executável, sem mudar PATH/configuração global da máquina.
  add_custom_command(TARGET factory_tests POST_BUILD COMMAND "${CMAKE_COMMAND}"
    -E copy_if_different "$<TARGET_FILE:driver_mediapipe_hands>" "$<TARGET_FILE_DIR:factory_tests>")
endif()
add_test(NAME factory COMMAND factory_tests)
add_executable(wire_probe tests/wire_probe.cpp)
add_test(NAME python_native_wire COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/wire_integration.py" "$<TARGET_FILE:wire_probe>")
set_tests_properties(protocol receiver factory python_native_wire PROPERTIES TIMEOUT 10)
